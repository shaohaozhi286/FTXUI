19bool IsSpaceGlyph(const std::string& glyph) {
20 return glyph == " " || glyph == "\t";
21}
22
23bool IsNarrowGlyph(const std::string& glyph) {
24 return string_width(glyph) == 1;
25}

27bool IsOpeningCjkPunctuation(const std::string& glyph) {
28 static constexpr std::array<std::string_view, 8> kOpening = {
29 "（", "《", "「", "【", "‘", "“", "〈", "『"};
30 for (const auto& candidate : kOpening) {
31 if (glyph == candidate) {
32 return true;
33 }
34 }
35 return false;
36}
37
38bool IsClosingCjkPunctuation(const std::string& glyph) {
39 static constexpr std::array<std::string_view, 15> kClosing = {
40 "，", "。", "！", "？", "；", "：", "、", "）",
41 "》", "」", "】", "’", "”", "〉", "』"};
42 for (const auto& candidate : kClosing) {
43 if (glyph == candidate) {
44 return true;
45 }
46 }
47 return false;
48}

50void FlushAsciiRun(std::string* ascii_run,
51 std::string* pending_prefix,
52 std::vector<std::string>* output) {
53 if (ascii_run->empty()) {
54 return;
55 }
56
57 std::string token = std::move(*ascii_run);
58 ascii_run->clear();
59 if (!pending_prefix->empty()) {
60 token = std::move(*pending_prefix) + token;
61 pending_prefix->clear();
62 }
63 output->push_back(std::move(token));
64}

66Elements Split(std::string_view the_text) {
67 std::vector<std::string> tokens;
68 std::string ascii_run;
69 std::string pending_prefix;
70
71 for (const auto& glyph : Utf8ToGlyphs(the_text)) {
72 if (glyph.empty()) {
73 continue;
74 }
75
76 if (IsSpaceGlyph(glyph)) {
77 FlushAsciiRun(&ascii_run, &pending_prefix, &tokens);
78 if (!tokens.empty()) {
79 tokens.back() += " ";
80 }
81 continue;
82 }
83
84 if (IsOpeningCjkPunctuation(glyph)) {
85 FlushAsciiRun(&ascii_run, &pending_prefix, &tokens);
86 pending_prefix += glyph;
87 continue;
88 }
89
90 if (IsClosingCjkPunctuation(glyph)) {
91 FlushAsciiRun(&ascii_run, &pending_prefix, &tokens);
92 if (!tokens.empty()) {
93 tokens.back() += glyph;
94 } else if (!pending_prefix.empty()) {
95 pending_prefix += glyph;
96 } else {
97 tokens.push_back(glyph);
98 }
99 continue;
100 }
101
102 if (IsNarrowGlyph(glyph)) {
103 ascii_run += glyph;
104 continue;
105 }
106
107 FlushAsciiRun(&ascii_run, &pending_prefix, &tokens);
108 std::string token = glyph;
109 if (!pending_prefix.empty()) {
110 token = std::move(pending_prefix) + token;
111 pending_prefix.clear();
112 }
113 tokens.push_back(std::move(token));
114 }
115
116 FlushAsciiRun(&ascii_run, &pending_prefix, &tokens);
117 if (!pending_prefix.empty()) {
118 if (!tokens.empty()) {
119 tokens.back() += pending_prefix;
120 } else {
121 tokens.push_back(std::move(pending_prefix));
122 }
123 }
124
125 Elements output;
126 output.reserve(tokens.size());
127 for (auto& token : tokens) {
128 output.push_back(text(std::move(token)));
129 }
130
131 return output;
132}

134Element Split(std::string_view paragraph,
135 const std::function<Element(std::string_view)>& f) {
136 Elements output;
137 size_t start = 0;
138 size_t end = paragraph.find('\n');
139 while (end != std::string_view::npos) {
140 output.push_back(f(paragraph.substr(start, end - start)));
141 start = end + 1;
142 end = paragraph.find('\n', start);
143 }
144 output.push_back(f(paragraph.substr(start)));
145 return vbox(std::move(output));
146}

157/// @brief Return an element drawing the paragraph on multiple lines, aligned on
158/// the left.
159/// @ingroup dom
160/// @see flexbox.
161Element paragraphAlignLeft(std::string_view the_text) {
162 return Split(the_text, [](std::string_view line) {
163 static const auto config = FlexboxConfig().SetGap(0, 0);
164 return flexbox(Split(line), config);
165 });
166};

174 static const auto config = FlexboxConfig().SetGap(0, 0).Set(

192/// @brief Return an element drawing the paragraph on multiple lines, aligned
193/// using a justified alignment.
194/// the center.
195/// @ingroup dom
196/// @see flexbox.
197Element paragraphAlignJustify(std::string_view the_text) {
198 return Split(the_text, [](std::string_view line) {
199 static const auto config = FlexboxConfig().SetGap(0, 0).Set(
200 FlexboxConfig::JustifyContent::SpaceBetween);
201 Elements words = Split(line);
202 words.push_back(text("") | xflex);
203 return flexbox(std::move(words), config);
204 });
205}
